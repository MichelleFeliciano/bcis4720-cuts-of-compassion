using System;
using System.Collections.Generic;
using System.Data.SqlClient;
using System.Linq;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace CutsofCompassion
{

   public partial class delete : System.Web.UI.Page
  {

     protected void Page_Load(object sender, EventArgs e)
     {

        if (Session["user"] == null)
        {
           // Redirect to the login page if the user is not logged in
           Response.Redirect("default.aspx");
        }
     }

     protected void DeleteButton_Click(object sender, EventArgs e)
     {

        string userEmail = Session["Email"]?.ToString(); // Retrieve email from session
        string enteredEmail = DEmailTextBox.Text;

        if (string.IsNullOrEmpty(userEmail) || string.IsNullOrEmpty(enteredEmail))
        {

           // Handle empty values gracefully
           Response.Write("<script>alert('Email is missing. Please try again.');</script>");
           return;
        }

        if (userEmail.Equals(enteredEmail, StringComparison.OrdinalIgnoreCase))
        {

           try
           {

              using (SqlConnection dcon = new
SqlConnection(SqlDataSource1.ConnectionString))

              {
                 using (SqlCommand dcommand = new

SqlCommand(SqlDataSource1.DeleteCommand, dcon))
                 {
                    dcommand.Parameters.AddWithValue("@Email", userEmail);
                    dcon.Open();
                    int rowsAffected = dcommand.ExecuteNonQuery();
                    dcon.Close();

                    if (rowsAffected > 0)
                    {

                       Response.Redirect("Success.aspx");
                    }
                    else
                    {

                       Response.Write("<script>alert('No account found to
delete.');</script>");

                    }
                 }
              }
           }
           catch (Exception ex)
           {
              // Log error (in production use a logging framework)
              Response.Write("<script>alert('An error occurred: " + ex.Message +
"');</script>");
           }
        }
        else
        {
           Response.Write("<script>alert('The email entered does not match your
account.');</script>");

        }
     }
  }
}

using System;

using System.Data.SqlClient;

using System.Linq;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace CutsofCompassion
{

   public partial class update : System.Web.UI.Page
  {

     protected void Page_Load(object sender, EventArgs e)
     {

        if (Session["User"] == null)
        {

           Response.Redirect("default.aspx");
        }

        if (!IsPostBack)
        {

           // Initially hide all textboxes and labels
           txtPass.Visible = false;
           txtCPass.Visible = false;
           txtFname.Visible = false;
           txtLname.Visible = false;
           txtYOB.Visible = false;
           // Declare variables for storing data from the database
           string sEmail = "";
           string sPass = "";
           string sFname = "";
           string sLname = "";
           string sYOB = "";

           // SQL Connection and Command to fetch data based on Email
           SqlDataReader myReader;
           using (SqlConnection myConnection = new
SqlConnection(SqlDataSource1.ConnectionString))
           {

              // SQL Command to select user by Email
              string query = "SELECT Email, Password, FName, LName, YOB FROM
dbo.Login WHERE Email = @Email"; // Assuming the schema is dbo
              using (SqlCommand myCommand = new SqlCommand(query,
myConnection))
              {

                 myCommand.Parameters.AddWithValue("@Email", EmailLabel.Text);

                 // Open the database connection
                 myConnection.Open();
                 myReader =
myCommand.ExecuteReader(CommandBehavior.CloseConnection);
// Read the data from the database
while (myReader.Read())
{

   sEmail = myReader.GetString(0);
   sPass = myReader.GetString(1);
   sFname = myReader.GetString(2);
   sLname = myReader.GetString(3);
   sYOB = myReader.GetInt32(4).ToString();

   // Populate labels with the data
   EmailLabel.Text = sEmail;
   PassLabel.Text = sPass;
   FNLabel.Text = sFname;
   LNLabel.Text = sLname;
  YOBLabel.Text = sYOB;

   // Populate textboxes only if the corresponding checkbox is checked
   if (PassCheckBox.Checked)
   {

      txtPass.Text = sPass;
   }

   if (FNCheckBox.Checked)
   {

      txtFname.Text = sFname;
                    }

                    if (LNCheckBox.Checked)
                    {

                       txtLname.Text = sLname;
                    }

                    if (YOBCheckBox.Checked)
                    {

                       txtYOB.Text = sYOB;
                    }
                 }
              }
           }
        }
     }

     protected void BtnuUpdate_Click(object sender, EventArgs e)
     {

        // Declare a connection to the database
        using (SqlConnection UConnection = new
SqlConnection(SqlDataSource1.ConnectionString))
        {

           string updateQuery = "UPDATE Users SET Password = @Password, FirstName
= @Fname, LastName = @Lname, YOB = @YOB WHERE Email = @Email";

           using (SqlCommand UCommand = new SqlCommand(updateQuery,
UConnection))
{
  // Set the email parameter
  UCommand.Parameters.AddWithValue("@Email", EmailLabel.Text);

  // Check if the checkbox for each field is checked and add parameters
  if (PassCheckBox.Checked == true)
  {

     UCommand.Parameters.AddWithValue("@Password", txtPass.Text);
  }
  else
  {

     // Use the previous password value if checkbox is not checked
     UCommand.Parameters.AddWithValue("@Password", PassLabel.Text);
  }

  if (FNCheckBox.Checked == true)
  {

     UCommand.Parameters.AddWithValue("@Fname", txtFname.Text);
  }
  else
  {

     // Use the previous first name value if checkbox is not checked
     UCommand.Parameters.AddWithValue("@Fname", FNLabel.Text);
  }

  if (LNCheckBox.Checked == true)
{
  UCommand.Parameters.AddWithValue("@Lname", txtLname.Text);

}
else
{

  // Use the previous last name value if checkbox is not checked
  UCommand.Parameters.AddWithValue("@Lname", LNLabel.Text);
}

if (YOBCheckBox.Checked == true)
{

  UCommand.Parameters.AddWithValue("@YOB", txtYOB.Text);
}
else
{

  // Use the previous YOB value if checkbox is not checked
  UCommand.Parameters.AddWithValue("@YOB", YOBLabel.Text);
}

try
{

  // Open the connection and execute the command
  UConnection.Open();
  UCommand.ExecuteNonQuery();
  UConnection.Close();
        }
        finally
        {

           // Redirect to another page after updating
           Response.Redirect("default.aspx");
        }
     }
  }
}
